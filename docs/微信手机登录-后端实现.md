**方法**: `POST`
**请求体**:
```json

  "code": "微信登录code",
  "encrypted_data": "加密的手机号数据",
  "iv": "解密算法初始向量"
```


### 步骤 2: 创建微信解密工具

```python
# backend/app/utils/wechat_decrypt.py
from Crypto.Cipher import AES
import base64

class WeChatDecrypt:
    def __init__(self, appid, session_key):
        self.appid = appid
        self.session_key = session_key
    
    def decrypt_phone(self, encrypted_data, iv):
        """解密微信手机号数据"""
        session_key = base64.b64decode(self.session_key)
        encrypted_data = base64.b64decode(encrypted_data)
        iv = base64.b64decode(iv)
        
        cipher = AES.new(session_key, AES.MODE_CBC, iv)
        decrypted = cipher.decrypt(encrypted_data)
        
        # 去除补位
        pad = decrypted[-1]
        decrypted = decrypted[:-pad]
        
        data = json.loads(decrypted.decode('utf-8'))
        
        # 验证 appId
        if data['watermark']['appid'] != self.appid:
            raise ValueError('Invalid appId')
        
        return {
            'phone': data['phoneNumber'],
            'pure_phone': data['purePhoneNumber'],
            'country_code': data['countryCode']
        }
```

### 步骤 3: 实现接口

```python
# backend/app/routers/auth.py

from app.utils.wechat_decrypt import WeChatDecrypt
import requests

@router.post("/wechat-phone-login")
async def wechat_phone_login(
    request: WeChatPhoneLoginRequest,
    db: AsyncSession = Depends(get_db)
):
    """
    微信手机号授权登录
    
    1. 使用 code 获取 session_key
    2. 解密手机号数据
    3. 创建/更新用户
    4. 返回用户信息和 session
    """
    # 1. 获取 session_key
    wx_url = f"https://api.weixin.qq.com/sns/jscode2session"
    params = {
        "appid": settings.wechat_miniapp_appid,
        "secret": settings.wechat_miniapp_secret,
        "js_code": request.code,
        "grant_type": "authorization_code"
    }
    
    wx_response = requests.get(wx_url, params=params)
    wx_data = wx_response.json()
    
    if wx_data.get('errcode'):
        raise HTTPException(400, f"微信登录失败: {wx_data['errmsg']}")
    
    session_key = wx_data['session_key']
    openid = wx_data['openid']
    
    # 2. 解密手机号
    decryptor = WeChatDecrypt(settings.wechat_miniapp_appid, session_key)
    phone_data = decryptor.decrypt_phone(request.encrypted_data, request.iv)
    
    phone = phone_data['pure_phone']
    
    # 3. 创建或查找用户
    user = await UserService.get_or_create_by_phone(db, phone, openid)
    
    # 4. 创建 session
    session = await SessionService.create_session(db, user.id)
    
    return {
        "id": user.id,
        "phone": user.phone,
        "is_guest": False,
        "token": session.id
    }
```
